## 🚀 WHAT YOU CAN DO NOW

### 1. Use AI Agents

```python
from agents.base import BaseAgent
from agents.communication import MessageBus, Blackboard

# Create communication infrastructure
message_bus = MessageBus()
blackboard = Blackboard()

# Create custom agent
class TradingAgent(BaseAgent):
    async def perceive(self, environment):
        return {'market_data': environment['candles']}
    
    async def reason(self, perception):
        return {'signal': 'CALL', 'confidence': 0.85}
    
    async def act(self, decision):
        await self.write_to_blackboard('signal', decision)
        return {'action': 'signal_published'}

# Use agent
agent = TradingAgent("trader", "trading", message_bus=message_bus, blackboard=blackboard)
result = await agent.run_cycle(environment)
```

### 2. Use OpenClaw

```python
from ai.models.openclaw_model import create_openclaw_model

# Create model (defaults to Ollama + Llama 3.2)
model = create_openclaw_model()

# Get prediction
prediction = model.predict(market_data)
print(f"Signal: {prediction['signal']}")
print(f"Confidence: {prediction['confidence']}%")
```

### 3. Use Consensus with OpenClaw

```python
from ai.models.openclaw_model import OpenClawModel
from ai.models.consensus_engine import AIConsensusEngine

# Create consensus

consensus.add_model(OpenClawModel(backend="ollama", model_name="mistral"))
```
